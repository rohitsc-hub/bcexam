function isHappy(n) {
    let seen = new Set();
    while (n !== 1 && !seen.has(n)) {
        seen.add(n);
        let sum = 0;
        while (n > 0) {
            let digit = n % 10;
            sum += digit * digit;
            n = Math.floor(n / 10);
        }
        n = sum;
    }
    return n === 1;
}

armstrong.py
import hashlib
import datetime
class Block:
    def __init__(self, i, n, ph):
        self.i, self.n = i, n
        self.t = str(datetime.datetime.now())
        self.r = "Armstrong number" if n == sum(int(d)**len(str(n)) for d in str(n)) else "not Armstrong number"
        self.ph = ph
        self.h = self.calhash()
    def calhash(self):
        data = str(self.i) + self.t + str(self.n) + self.r + self.ph
        return hashlib.sha256(data.encode()).hexdigest()
class Blockchain:
    def __init__(self):
        self.chain = [Block(0, 0, "0")]
    def addBlock(self, n):
        self.chain.append(Block(len(self.chain), n, self.chain[-1].h))
    def display(self):
        for b in self.chain:
            print("Block:", b.i)
            print("Timestamp:", b.t)
            print("Original number:", b.n)
            print("Result:", b.r)
            print("Previous hash:", b.ph)
            print("Current Hash:", b.h)
            print("-" * 95)
    def is_valid(self):
        return all(
            b.h == b.calhash() and b.ph == self.chain[i-1].h
            for i, b in enumerate(self.chain[1:], 1)
        )
p = Blockchain()
for n in [12, 153, 9563]:
    p.addBlock(n)
p.display()

import datetime

    def __init__(s, i, str1, str2, t, ph):
        s.i = i
        s.t = str(datetime.datetime.now())
        s.str1 = str1
        s.str2 = str2
        s.r = s.check()
        s.ph = ph
        s.h = s.calhash()
    def calhash(s):
        data = str(s.i) + s.t + s.str1 + s.str2 + s.r + s.ph
        return hashlib.sha256(data.encode()).hexdigest()
    def check(s):
        a = s.str1.replace(" ", "").lower()

import hashlib
import datetime
class Block:
    def __init__(s, i, acc, act, amt, ph):
        s.i, s.acc, s.act, s.amt, s.ph = i, acc, act, amt, ph
        s.t = str(datetime.datetime.now())
        s.h = hashlib.sha256(
            (str(i)+s.t+acc+act+str(amt)+ph).encode()
        ).hexdigest()
class Blockchain:
    def __init__(s):
        s.chain = [Block(0, "GENESIS", "Create", 0, "0")]
        s.accounts = {}
    def add(s, acc, act, amt):
        if act == "Account Created":
            if acc in s.accounts:
                print("Account already exists")
                return
            s.accounts[acc] = 0
            print("Account created successfully")
        elif acc not in s.accounts:
            print("Account does not exist")
            return
        elif act == "Deposit":
            s.accounts[acc] += amt
            print("Deposit successful")
            print("Balance:", s.accounts[acc])
        elif act == "Withdraw":
            if amt > s.accounts[acc]:
                print("Insufficient balance")
                return
            s.accounts[acc] -= amt
            print("Withdrawal successful")
            print("Balance:", s.accounts[acc])
        s.chain.append(Block(len(s.chain), acc, act, amt, s.chain[-1].h))
    def display(s):
        for b in s.chain:
            print("Block:", b.i)
            print("Account:", b.acc)
            print("Action:", b.act)
            print("Amount:", b.amt)
            print("Timestamp:", b.t)
            print("Previous Hash:", b.ph)
            print("Current Hash:", b.h)
            print("-" * 95)
    def balance(s, acc):
        if acc in s.accounts:
            print("Account:", acc)
            print("Balance:", s.accounts[acc])
        else:
            print("Account does not exist")

contract voting {
    string[] public candidates;
    constructor(string[] memory names) {
        candidates = names;
    }


    mapping(string => uint) public countvotes;
    mapping(address => bool) public voterAddress;


    function vote(string memory c) public {
        require(!voterAddress[msg.sender], "you have alredy voted");
        bool ce = false;
        for (uint i = 0; i < candidates.length; i++) {
            if (keccak256(bytes(c)) == keccak256(bytes(candidates[i]))) {
                ce = true;
                break;
            }
        }
        require(ce, "Condidate dows not exists");
        countvotes[c]++;
        voterAddress[msg.sender] = true;
    }


    function getWinner() public view returns (string memory) {
        string memory w;
        uint t = 0;
        for (uint i = 0; i < candidates.length; i++) {
            if (countvotes[candidates[i]] > t) {
                t = countvotes[candidates[i]];
                w = candidates[i];
            }
        }
        return w;
    }
}

Q.2. Write a Solidity smart contract using a Student structure to store student details and marks, and implement functions to add students, count students, display student details, calculate grades and retrieve the details of a particular student.




Description: The smart contract StudentGrade defines a Student structure containing the student's ID, name, address, age, and an array of marks. A mapping stud_id associates each student ID with its corresponding student details.
The addStudent() function adds a new student and prevents duplicate student IDs using require(). The studentIds array stores the IDs separately so that all students can be accessed using a loop, while studentCount keeps track of the total number of students.
The totalStudent() function returns the total number of students. The getAllStudents() function returns the details of all students. The getGrade() function calculates the average of the student's marks and returns a grade based on the average marks. The displayDetail() function returns the complete details of a particular student using the student's ID.
The contract is written using Solidity version 0.8.21 and can be compiled and deployed using the Remix IDE.


Student.sol
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.21;


contract StudentGrade {
    struct Student {
        uint id;
        string name;
        string address_;
        uint age;
        uint[] marks;
    }


    Student[] public students;


    function addStudent(uint id, string memory name, string memory addr, uint age, uint[] memory marks) public {
        students.push(Student(id, name, addr, age, marks));
    }


    function displayStudent() public view returns (Student[] memory) {
        return students;
    }


    function countStudents() public view returns (uint) {
        return students.length;
    }


    function getGrade(uint id) public view returns (string memory) {
        for (uint i = 0; i < students.length; i++) {
            if (students[i].id == id) {
                uint total = 0;


                if (students[i].marks.length == 0)
                    return "No marks";


                for (uint j = 0; j < students[i].marks.length; j++)
                    total += students[i].marks[j];


                uint avg = total / students[i].marks.length;


                if (avg >= 90) return "Grade: A+";
                if (avg >= 80) return "Grade: A";
                if (avg >= 70) return "Grade: B";
                if (avg >= 60) return "Grade: C";
                if (avg >= 50) return "Grade: D";
                return "Grade: F";
            }
        }
        return "Student not found";
    }


    function getStudent(uint id) public view returns (uint, string memory, string memory, uint, uint[] memory) {
        for (uint i = 0; i < students.length; i++) {
            if (students[i].id == id) {
                Student memory s = students[i];
                return (s.id, s.name, s.address_, s.age, s.marks);
            }
        }
        revert("Student not found");
    }
}

Q.2 Crowdfunding Contract
Create a crowdfunding contract with an owner and a funding goal.
Requirements:
* Initialize the owner and funding goal using a constructor.
* Let users contribute ETH.
* Emit a ContributionReceived event for every contribution.
* Emit a GoalReached event when the total reaches the goal.
* Allow refunds if the campaign fails after its deadline.
* Use a parent contract for campaign details and a child contract for contribution logic.
Funding.sol
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.21;


contract Crowdfunding {
    address public owner;
    uint public fundingGoal;
    uint public deadline;
    uint public totalFunds;


    event ContributionReceived(address, uint);
    event GoalReached(uint);


    constructor(uint _fundingGoal, uint _duration) {
        owner = msg.sender;
        fundingGoal = _fundingGoal;
        deadline = block.timestamp + _duration;
    }


    function dispo() public view returns (address) {
        return owner;
    }
}


contract CrowdfundingCampaign is Crowdfunding {


    constructor(uint _fundingGoal, uint _duration)
        Crowdfunding(_fundingGoal, _duration) {}


    function contribute() public payable {
        require(block.timestamp < deadline, "Campaign ended");
        require(msg.value > 0, "Invalid contribution");


        totalFunds += msg.value;
        emit ContributionReceived(msg.sender, msg.value);


        if (totalFunds >= fundingGoal) {
            emit GoalReached(totalFunds);
        }
    }


    function refund() public payable {
        require(block.timestamp >= deadline, "Campaign not ended");
        require(totalFunds < fundingGoal, "Goal reached");


        uint amount = address(this).balance;
        payable(msg.sender).transfer(amount);
    }
}

            throw new Error("Contract not found. Deploy again.");
        const contract = new ethers.Contract(contractAddress, ABI, provider);
        const result = await contract.checkPerfectNumber(BigInt(n));
        output.innerText = result ? "Perfect Number" : "Not a Perfect Number";
    } catch (error) {
        console.error(error);
        output.innerText = error.shortMessage || error.message;
    }
});

const c = require('crypto');
class Block {
    constructor(id, n1, t, ph = "") { this.id = id;
        this.n1 = n1;
        this.t = t;
        this.ans = this.checkArmstrong();
        this.ph = ph;
        this.h = this.calHash();
    }
    checkArmstrong() {
        let n = this.n1;
        let digits = String(n).length;
        let temp = n;
        let sum = 0;
        if (n < 0) return "Not Armstrong";
        while (temp > 0) {
            let r = temp % 10;
            sum += r ** digits;
            temp = Math.floor(temp / 10);
        }
        return sum == n ? "Armstrong" : "Not Armstrong";
    }
    calHash() {
        return c.createHash('sha256').update(this.id + this.t + this.n1 + this.ans + this.ph).digest('hex');
    }
}

        return this.chain[this.chain.length - 1];
    }
    addBlock(b) {
        b.ph = this.getCblock().h;
        b.h = b.calHash();
        this.chain.push(b);
    }
}

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

contract Krishnamurthy {
    function checkKrishnamurthy(uint n) public pure returns (bool) {
        uint temp = n;
        uint sum = 0;

        if (n == 0) return false;
        while (temp > 0) {
            uint r = temp % 10;
            uint fact = 1;
            for (uint i = 1; i <= r; i++) {
                fact *= i;
            }
            sum += fact;
            temp = temp / 10;
        }
        return sum == n;
    }
}
